qemu-system-x86_64만 cmd에 입력시 실행이 된다.
nasm -f bin boot12.asm -o boot12.bin
copy /b boot12.bin os12.img
qemu-system-x86_64 -drive format=raw,file=os12.img

커널 빌드하는 방식 -> vga파일도 컴파일을 해야 한다.
gcc -m32 -ffreestanding -fno-pie -fno-stack-protector -c kernel.c -o kernel.o
gcc -m32 -ffreestanding -fno-pie -fno-stack-protector -c vga.c -o vga.o
//키보드 파일 컴파일
gcc -m32 -ffreestanding -fno-pie -fno-stack-protector -c keyboard.c -o keyboard.o

nasm -f elf32 kernel_entry.asm -o kernel_entry.o

//새로 추가된 함수를 object code로 만드는 코드다.
nasm -f elf32 lowlevel.asm -o lowlevel.o

nasm -f elf32 interrupts.asm -o interrupts.o

nasm -f elf32 interrupts.asm -o interrupts.o

ld -m elf_i386 -T linker.ld kernel_entry.o kernel.o vga.o lowlevel.o interrupts.o keyboard.o -o kernel.elf

objcopy -O binary kernel.elf kernel.bin
패딩을 채워넣는 방식
cp kernel.bin kernel_padded.bin
truncate -s 16384 kernel_padded.bin

os img다시 채워넣기
nasm -f bin boot12.asm -o boot12.bin
cat boot12.bin kernel_padded.bin > os12.img
